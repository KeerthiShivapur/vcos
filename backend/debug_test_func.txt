async def compute_detailed(self) -> Dict[str, Any]:
        """Compute detailed fingerprint breakdown for comparison."""
        fingerprint = await self.compute()
        
        # Also return counts for each domain
        counts = {}
        
        org = await self.db.get(Organization, self.organization_id)
        counts["organization"] = 1 if org else 0
        
        for model_name, model_class, filter_field in [
            ("employees", Employee, "organization_id"),
            ("departments", Department, "organization_id"),
            ("teams", Team, "organization_id"),
            ("roles", Role, "organization_id"),
            ("decisions", CompanyDecision, "organization_id"),
            ("commitments", CompanyCommitment, "organization_id"),
            ("authorizations", CompanyAuthorization, "organization_id"),
            ("actions", CompanyAction, "organization_id"),
            ("timeline_events", CompanyTimelineEvent, "organization_id"),
            ("knowledge_snapshots", CompanyKnowledgeSnapshot, "organization_id"),
            ("autonomy_proposals", AutonomyProposal, "organization_id"),
            ("approval_policies", ApprovalPolicy, "organization_id"),
        ]:
            result = await self.db.execute(
                select(func.count(model_class.id))
                .where(getattr(model_class, filter_field) == self.organization_id)
            )
            counts[model_name] = result.scalar() or 0
        
        return {
            "fingerprint": fingerprint,
            "organization_id": self.organization_id,
            "computed_at": utcnow().isoformat(),
            "counts": counts,
        }
